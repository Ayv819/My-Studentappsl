<script>
let currentGroup = localStorage.getItem("currentGroup") || "General Chat";
let mediaRecorder;
let audioChunks = [];

document.getElementById("currentGroup").innerText = "You are in: " + currentGroup;
loadMessages();

function joinGroup(name){
  currentGroup = name;
  localStorage.setItem("currentGroup", name);
  document.getElementById("currentGroup").innerText = "You are in: " + name;
  loadMessages();
}

function getKey(){ return "chat_" + currentGroup; }

function loadMessages(){
  let box = document.getElementById("messages");
  box.innerHTML = "";
  let msgs = JSON.parse(localStorage.getItem(getKey()) || "[]");
  msgs.forEach(m => renderMessage(m));
  box.scrollTop = box.scrollHeight;
}

function renderMessage(m){
  let box = document.getElementById("messages");
  let div = document.createElement("div");
  div.className = "msg me";
  let content = "";
  if(m.type === "text") content = m.text;
  if(m.type === "image") content = '<img src="'+m.data+'">';
  if(m.type === "video") content = '<video controls src="'+m.data+'"></video>';
  if(m.type === "audio") content = '<audio controls src="'+m.data+'"></audio>';
  if(m.type === "file") content = '📄 File shared: ' + m.name;
  div.innerHTML = "<small>You</small><br>" + content;
  box.appendChild(div);
}

function saveMessage(msg){
  let msgs = JSON.parse(localStorage.getItem(getKey()) || "[]");
  msgs.push(msg);
  localStorage.setItem(getKey(), JSON.stringify(msgs));
  renderMessage(msg);
  document.getElementById("messages").scrollTop = 99999;
}

function sendText(){
  let input = document.getElementById("textInput");
  if(input.value.trim() === "") return;
  saveMessage({type:"text", text:input.value});
  input.value = "";
}

// ATTACH FILE
document.getElementById("fileInput").addEventListener("change", function(e){
  let file = e.target.files[0];
  if(!file) return;
  let reader = new FileReader();
  reader.onload = function(){
    let type = "file";
    if(file.type.startsWith("image")) type = "image";
    if(file.type.startsWith("video")) type = "video";
    saveMessage({type:type, data:reader.result, name:file.name});
  };
  reader.readAsDataURL(file);
  e.target.value = "";
});

// AUDIO RECORDING
document.getElementById("micBtn").addEventListener("click", async function(){
  let btn = this;
  if(!mediaRecorder || mediaRecorder.state === "inactive"){
    try{
      let stream = await navigator.mediaDevices.getUserMedia({audio:true});
      mediaRecorder = new MediaRecorder(stream);
      audioChunks = [];
      mediaRecorder.ondataavailable = e => audioChunks.push(e.data);
      mediaRecorder.onstop = () => {
        let blob = new Blob(audioChunks, {type:"audio/webm"});
        let reader = new FileReader();
        reader.onload = () => saveMessage({type:"audio", data:reader.result});
        reader.readAsDataURL(blob);
        stream.getTracks().forEach(t => t.stop());
      };
      mediaRecorder.start();
      btn.innerText = "⏹ Stop & Send";
      document.getElementById("recordStatus").innerText = "Recording... speak now";
    }catch(err){
      alert("Microphone permission denied. Please allow mic access.");
    }
  } else {
    mediaRecorder.stop();
    btn.innerText = "🎤 Record Audio";
    document.getElementById("recordStatus").innerText = "";
  }
});
</script>
